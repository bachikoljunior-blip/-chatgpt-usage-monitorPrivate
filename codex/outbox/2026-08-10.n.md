    (() => {
      const STORAGE_KEY = 'hoshikuzu-kobo-v2';
      const LEGACY_KEY = 'hoshikuzu-kobo-v1';

      const PATHS = {
        burst: {
          name: '瞬間派（タップ強化）',
          tap: 2.0,
          rate: 1.0,
          cost: 1.18
        },
        steady: {
          name: '持続派（自動強化）',
          tap: 1.0,
          rate: 1.45,
          cost: 0.92
        }
      };

      const UNIT_DEFS = [
        { key: 'harvester', name: '星核ハーベスタ', baseRate: 0.35, baseCost: 4 },
        { key: 'satellite', name: '衛星結線塔', baseRate: 1.2, baseCost: 120 },
        { key: 'reactor', name: '重力反応炉', baseRate: 6.0, baseCost: 760 }
      ];

      const el = {
        pathName: document.getElementById('pathName'),
        runCount: document.getElementById('runCount'),
        totalDust: document.getElementById('totalDust'),
        sessionDust: document.getElementById('sessionDust'),
        dust: document.getElementById('dust'),
        perSec: document.getElementById('perSec'),
        statusLine: document.getElementById('statusLine'),
        unitList: document.getElementById('unitList'),
        overlay: document.getElementById('pathOverlay'),
        tapBtn: document.getElementById('tapBtn'),
        saveInfo: document.getElementById('saveInfo'),
        resetRun: document.getElementById('resetRun')
      };

      function freshState() {
        return {
          version: 2,
          path: null,
          pathChosenAt: null,
          runCount: 1,
          totalDust: 0,
          totalTouched: 0,
          totalSpent: 0,
          dust: 0,
          runDust: 0,
          runTapped: 0,
          lastTick: Date.now(),
          units: UNIT_DEFS.map(() => ({ owned: 0 })),
          lastSave: Date.now()
        };
      }

      function numberAtLeast(v, fallback = 0) {
        return Number.isFinite(v) ? v : fallback;
      }

      function loadJson(key) {
        const raw = localStorage.getItem(key);
        if (!raw) return null;
        try {
          const parsed = JSON.parse(raw);
          if (!parsed || typeof parsed !== 'object') return null;
          return parsed;
        } catch {
          return null;
        }
      }

      function normalizeState(src) {
        if (!src || typeof src !== 'object') return freshState();

        const s = freshState();
        if (src.version === 1 || src.version === 2 || src.version === undefined) {
          if (src.version) s.version = numberAtLeast(src.version, 2);
          if (typeof src.path === 'string' && PATHS[src.path]) s.path = src.path;
          if (typeof src.runCount === 'number') s.runCount = Math.max(1, Math.floor(src.runCount));
          if (typeof src.totalDust === 'number') s.totalDust = Math.max(0, src.totalDust);
          if (typeof src.totalTouched === 'number') s.totalTouched = Math.max(0, Math.floor(src.totalTouched));
          if (typeof src.totalSpent === 'number') s.totalSpent = Math.max(0, src.totalSpent);
          if (typeof src.dust === 'number') s.dust = Math.max(0, src.dust);
          if (typeof src.runDust === 'number') s.runDust = Math.max(0, src.runDust);
          if (typeof src.runTapped === 'number') s.runTapped = Math.max(0, Math.floor(src.runTapped));
          if (typeof src.lastTick === 'number') s.lastTick = Math.max(0, src.lastTick);
          if (typeof src.pathChosenAt === 'number') s.pathChosenAt = src.pathChosenAt;
          if (typeof src.lastSave === 'number') s.lastSave = src.lastSave;
        }

        if (Array.isArray(src.units) && src.units.length === s.units.length) {
          s.units = s.units.map((u, i) => {
            const u2 = src.units[i];
            if (u2 && typeof u2 === 'object' && Number.isFinite(u2.owned)) {
              return { owned: Math.max(0, Math.floor(u2.owned)) };
            }
            return { owned: 0 };
          });
        }

        if (!s.path && typeof src.route === 'string' && PATHS[src.route]) {
          s.path = src.route;
        }
        if (!s.path && typeof src.selectedPath === 'string' && PATHS[src.selectedPath]) {
          s.path = src.selectedPath;
        }

        // 旧保存があれば、同じ値名のものを吸収（未知のキーを壊さない）
        if (typeof src.stardust === 'number' && s.totalDust === 0 && s.runDust === 0) {
          s.totalDust = Math.max(0, src.stardust);
          s.runDust = Math.max(0, src.stardust);
        }

        return s;
      }

      function loadState() {
        const next = loadJson(STORAGE_KEY);
        if (next) return normalizeState(next);

        const legacy = loadJson(LEGACY_KEY);
        if (legacy) return normalizeState(legacy);

        return freshState();
      }

      let state = loadState();
      if (!state.path) state.path = null;
      if (!state.lastTick || !Number.isFinite(state.lastTick)) state.lastTick = Date.now();
      if (typeof state.version !== 'number') state.version = 2;

      function currentPath() {
        return state.path && PATHS[state.path] ? PATHS[state.path] : null;
      }

      function applyOffline(now) {
        const path = currentPath();
        if (!path) return;
        const delta = Math.min(60 * 60 * 24, Math.max(0, (now - state.lastTick) / 1000));
        if (delta < 1) return;
        const gained = productionPerSecond() * delta;
        if (gained > 0) {
          const floorGain = gained;
          state.dust += floorGain;
          state.runDust += floorGain;
          state.totalDust += floorGain;
        }
        state.lastTick = now;
      }

      function productionPerSecond() {
        const path = currentPath();
        if (!path) return 0;
        return state.units.reduce((sum, u, idx) => {
          return sum + (u.owned * UNIT_DEFS[idx].baseRate * path.rate);
        }, 0);
      }

      function formatInt(v) {
        return Math.max(0, Math.floor(v || 0)).toLocaleString('ja-JP');
      }

      function formatFloat(v) {
        return Math.max(0, v || 0).toFixed(1);
      }

      function unitCost(index) {
        const path = currentPath();
        const base = UNIT_DEFS[index];
        const owned = state.units[index].owned;
        const costMul = path ? path.cost : 1;
        return Math.max(1, Math.floor(base.baseCost * Math.pow(1.15, owned) * costMul));
      }

      function unitRate(index) {
        const path = currentPath();
        const base = UNIT_DEFS[index];
        const rateMul = path ? path.rate : 1;
        return base.baseRate * rateMul;
      }

      function save() {
        state.lastSave = Date.now();
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      }

      function applyChoice(id) {
        if (!PATHS[id]) return;
        state.path = id;
        state.pathChosenAt = Date.now();
        state.lastTick = Date.now();
        save();
        render();
      }

      function canAfford(index) {
        return state.dust >= unitCost(index);
      }

      function exchange(index) {
        if (!state.path) return;
        const cost = unitCost(index);
        if (state.dust >= cost) {
          state.dust -= cost;
          state.units[index].owned += 1;
          state.totalSpent += cost;
          state.runDust += 0;
          save();
          render();
        }
      }

      function tap() {
        if (!state.path) return;
        const path = currentPath();
        const gain = 1 * path.tap;
        state.dust += gain;
        state.runDust += gain;
        state.totalDust += gain;
        state.totalTouched += 1;
        state.runTapped += 1;
        state.runDust = Math.max(state.runDust, 0);
        state.dust = Math.max(state.dust, 0);
        state.totalDust = Math.max(state.totalDust, 0);
        save();
        render();
      }

      function hardResetRun() {
        if (!window.confirm('この周回を初期化します。進路・通算値は残ります。')) return;
        state.runDust = 0;
        state.totalTouched += 0;
        state.runTapped = 0;
        state.lastTick = Date.now();
        state.units = UNIT_DEFS.map(() => ({ owned: 0 }));
        state.runCount += 1;
        save();
        render();
      }

      function refreshUI() {
        const path = currentPath();
        el.pathName.textContent = path ? path.name : '未選択';
        el.runCount.textContent = formatInt(state.runCount);
        el.totalDust.textContent = formatInt(state.totalDust);
        el.sessionDust.textContent = formatInt(state.runDust);
        el.dust.textContent = formatInt(state.dust);
        el.perSec.textContent = `${formatFloat(productionPerSecond())} /秒`;

        if (path) {
          el.statusLine.textContent = `タップ倍率 ${formatFloat(path.tap)}、自動倍率 ${formatFloat(path.rate)}、建設コスト補正 ${path.cost >= 1 ? (path.cost.toFixed(2) + '倍') : ((path.cost * 100).toFixed(0) + '%')}`;
        } else {
          el.statusLine.textContent = '進路を選ぶとその後の増え方が固定されます。';
        }

        el.overlay.classList.toggle('hidden', Boolean(state.path));
        el.tapBtn.disabled = !state.path;
        el.unitList.innerHTML = '';

        UNIT_DEFS.forEach((def, index) => {
          const owned = state.units[index].owned;
          const cost = unitCost(index);
          const row = document.createElement('div');
          row.className = 'unit';

          const left = document.createElement('div');
          left.className = 'unit-left';

          const title = document.createElement('div');
          title.className = 'unit-name';
          title.textContent = `${def.name} × ${owned}`;

          const meta = document.createElement('div');
          meta.className = 'unit-meta';
          meta.textContent = `一台あたり ${formatFloat(unitRate(index))} 粒/秒、コスト ${formatInt(cost)} 粒`;

          const right = document.createElement('div');
          right.className = 'unit-right';

          const btn = document.createElement('button');
          btn.className = 'btn';
          btn.textContent = '取得';
          btn.disabled = !state.path || !canAfford(index);
          btn.addEventListener('click', () => exchange(index));

          const gainHint = document.createElement('div');
          gainHint.className = 'small';
          gainHint.textContent = `建設後の合計増加: ${(unitRate(index) * owned).toFixed(1)} / 秒`;

          left.appendChild(title);
          left.appendChild(meta);
          right.appendChild(gainHint);
          right.appendChild(btn);

          row.appendChild(left);
          row.appendChild(right);
          el.unitList.appendChild(row);
        });

        if (!state.path) {
          setTimeout(() => {
            el.overlay.classList.remove('hidden');
          }, 10);
        }
      }

      function tick(now) {
        const path = currentPath();
        if (path) {
          applyOffline(now);

          const elapsed = Math.max(0, (now - state.lastTick) / 1000);
          if (elapsed >= 0.25) {
            const gain = productionPerSecond() * elapsed;
            state.dust += gain;
            state.runDust += gain;
            state.totalDust += gain;
            state.lastTick = now;
            save();
            refreshUI();
          }
        }
      }

      document.querySelectorAll('.choice-btn').forEach((btn) => {
        btn.addEventListener('click', () => {
          const id = btn.getAttribute('data-path');
          applyChoice(id);
        }, { passive: true });
      });

      el.tapBtn.addEventListener('click', () => {
        tap();
      }, { passive: true });

      el.saveInfo.addEventListener('click', () => {
        const loaded = loadState();
        state = loaded;
        if (state.path && !PATHS[state.path]) state.path = null;
        applyOffline(Date.now());
        refreshUI();
      });

      el.resetRun.addEventListener('click', hardResetRun, { passive: true });

      state.dust = numberAtLeast(state.dust);
      state.runDust = numberAtLeast(state.runDust);
      state.totalDust = numberAtLeast(state.totalDust);
      state.totalTouched = numberAtLeast(state.totalTouched, 0);
      state.totalSpent = numberAtLeast(state.totalSpent, 0);
      state.runTapped = numberAtLeast(state.runTapped, 0);

      applyOffline(Date.now());
      save();
      refreshUI();
      requestAnimationFrame(() => {
        tick(Date.now());
      });
      setInterval(() => {
        tick(Date.now());
      }, 250);
      window.addEventListener('beforeunload', save);
    })();
  </script>
