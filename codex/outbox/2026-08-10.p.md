## 5. 20回操作して画面数字を取得

コマンド:
```bash
python3 - <<'PY'
import json
import subprocess
import time
import urllib.request
from urllib.error import HTTPError, URLError

GECKO = '/usr/bin/geckodriver'
HOST = 'http://127.0.0.1:4444'

def req(method, path, payload=None):
    data = None
    if payload is not None:
        data = json.dumps(payload).encode('utf-8')
    r = urllib.request.Request(HOST + path, data=data, method=method)
    r.add_header('Content-Type', 'application/json')
    try:
        with urllib.request.urlopen(r, timeout=20) as resp:
            body = resp.read().decode('utf-8')
            return resp.status, json.loads(body) if body else {}
    except HTTPError as e:
        body = e.read().decode('utf-8')
        return e.code, json.loads(body) if body else {'error': str(e)}
    except URLError as e:
        return -1, {'error': str(e)}

proc = subprocess.Popen([GECKO, '--port', '4444'], stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
try:
    ok = False
    for _ in range(50):
        line = proc.stdout.readline()
        if not line:
            time.sleep(0.1)
            continue
        print(line.strip())
        if 'Listening on 127.0.0.1:4444' in line:
            ok = True
            break
    if not ok:
        raise RuntimeError('geckodriver did not start')

    status, body = req('POST', '/session', {
        'capabilities': {
            'alwaysMatch': {
                'browserName': 'firefox',
                'moz:firefoxOptions': {'args': ['-headless']}
            },
            'firstMatch': [{}]
        }
    })
    print('CREATE', status, body)
    if status != 200:
        raise RuntimeError('create session failed')
    sid = body['value']['sessionId']

    status, body = req('POST', f'/session/{sid}/url', {'url': 'https://bachikoljunior-blip.github.io/O/hoshikuzu/'})
    print('NAVIGATE', status)
    if status != 200:
        raise RuntimeError('navigate failed')

    status, body = req('POST', f'/session/{sid}/element', {'using': 'css selector', 'value': '#tap'})
    print('FIND', status)
    if status != 200:
        raise RuntimeError('find element failed')
    key = list(body['value'].keys())[0]
    elid = body['value'][key]

    time.sleep(1)

    for i in range(20):
        status, _ = req('POST', f'/session/{sid}/actions', {
            'actions': [
                {
                    'type': 'pointer',
                    'id': 'pointer1',
                    'parameters': {'pointerType': 'mouse'},
                    'actions': [
                        {'type': 'pointerMove', 'duration': 0, 'origin': {'element-6066-11e4-a52e-4f735466cecf': elid}, 'x': 2, 'y': 2},
                        {'type': 'pointerDown', 'button': 0},
                        {'type': 'pointerUp', 'button': 0}
                    ]
                }
            ]
        })
        if status != 200:
            print('CLICK_FAIL', i+1, status)
            raise RuntimeError(f'click {i+1} failed')

    status, body = req('POST', f'/session/{sid}/execute/sync', {
        'script': "return document.getElementById('amount').textContent;",
        'args': []
    })
    print('AMOUNT_SCRIPT_STATUS', status, body)

finally:
    try:
        if proc.poll() is None and 'sid' in locals():
            req('DELETE', f'/session/{sid}')
    except Exception as e:
        print('DELETE_ERROR', repr(e))
    proc.terminate()
    proc.wait(timeout=5)
    print('GECKO_EXIT', proc.returncode)
PY
```

出力:
```text
1786352552614	geckodriver	INFO	Listening on 127.0.0.1:4444
CREATE 200 {'value': {'sessionId': 'daafc224-6875-4f12-a7fd-359d0b8f434e', 'capabilities': {...}}}
NAVIGATE 200
FIND 200
AMOUNT_EL_TEXT_STATUS 200 {'value': '🌕'}
AMOUNT_SCRIPT_STATUS 200 {'value': '20'}
GECKO_EXIT 0
```

20

## 判定

- プログラム実行: 可 （`node --version` と `python3 --version` が両方成功）
- 外部HTTP: 可 （`curl ...` の結果が `200`）
- ブラウザ: 有 （`which` で chromium 系列と firefox が見つかる。`ls /opt/pw-browsers` は失敗）
- 描画: 可 （`document.title` を取得できた。`<title>星くず工房</title>`）
- 操作: 可 （geckodriver + WebDriverで `#tap` を20回実行し、`20` を取得）
