```javascript
// /tmp/run-game.js
const {spawn} = require('node:child_process');

const tmpPath = '/tmp/hoshikuzu-free-demo.html';
const base = 'http://127.0.0.1:9515';

async function httpJson(method, url, body) {
  const res = await fetch(url, {
    method,
    headers: { 'Content-Type': 'application/json' },
    body: body ? JSON.stringify(body) : undefined,
  });
  const txt = await res.text();
  let data = null;
  if (txt) {
    try { data = JSON.parse(txt); } catch (_) { data = txt; }
  }
  if (!res.ok) {
    const msg = typeof data === 'object' ? JSON.stringify(data) : txt;
    throw new Error(`HTTP ${res.status} ${res.statusText}: ${msg}`);
  }
  return data;
}

(async () => {
  const driver = spawn('chromedriver', ['--port=9515', '--url-base=/', '--allowed-origins=*']);

  const out = {stepLog: [], errors: []};

  try {
    await new Promise(r => setTimeout(r, 400));

    const create = await httpJson('POST', `${base}/session`, {
      desiredCapabilities: {
        browserName: 'chrome',
        'goog:chromeOptions': {
          args: ['--headless=new', '--no-sandbox', '--disable-gpu', '--disable-dev-shm-usage'],
        },
      },
    });

    const sessionId = create.sessionId || create.value?.sessionId;
    if (!sessionId) throw new Error('session not created');
    out.sessionId = sessionId;

    const exec = async (script, args = []) => {
      const r = await httpJson('POST', `${base}/session/${sessionId}/execute/sync`, { script, args });
      return r.value;
    };
    const call = async (path, body = {}) => {
      const r = await httpJson('POST', `${base}/session/${sessionId}${path}`, body);
      return r;
    };

    await call('/timeouts', { script: 30000, pageLoad: 30000, implicit: 0 });
    await call('/url', { url: `file://${tmpPath}` });

    const tapByPointer = async () => {
      await exec(`
        const el = document.querySelector('#tap');
        const evOpts = { bubbles: true, clientX: 12, clientY: 12 };
        el.dispatchEvent(new PointerEvent('pointerdown', evOpts));
        el.dispatchEvent(new PointerEvent('pointerup', evOpts));
      `);
    };

    // attach error capture
    await exec(`
      window.__consoleErrors = [];
      const pushErr = (...args) => { window.__consoleErrors.push(args.map(v => String(v)).join(' ')); };
      window.__consoleErrorHandler = (e) => { window.__consoleErrors.push('error:'+ (e && e.message ? e.message : String(e))); };
      window.addEventListener('error', window.__consoleErrorHandler);
      const orig = console.error;
      console.error = (...args) => { pushErr(...args); orig.apply(console, args); };
    `);

    const readCore = () => exec('return { amount: document.getElementById("amount").textContent, rate: document.getElementById("rate").textContent, goalText: document.getElementById("goalText").textContent };');
    const readAmount = () => exec('return document.getElementById("amount").textContent;');
    const readButtons = () => exec(`
      return Array.from(document.querySelectorAll('#shop .trade')).map((b, i) => ({
        index: i + 1,
        text: b.textContent.trim(),
        disabled: b.disabled,
        cost: b.querySelector('.cost')?.textContent || '',
        owned: b.querySelector('.owned')?.textContent || '',
      }));
    `);

    out.stepLog.push({ step: 1, ...await readCore() });

    for (let i = 0; i < 4; i++) {
      await tapByPointer();
    }
    out.stepLog.push({ step: 2, amount: await readAmount() });

    const shopInitial = await readButtons();
    out.stepLog.push({ step: 3, shop: shopInitial });

    const firstEnabled = await exec(`
      const buttons = Array.from(document.querySelectorAll('#shop .trade'));
      const idx = buttons.findIndex(b => !b.disabled);
      if (idx >= 0) buttons[idx].click();
      return idx;
    `);

    out.stepLog.push({
      step: 4,
      firstEnabled,
      ...await readCore(),
      shop: await readButtons(),
    });

    for (let i = 0; i < 10; i++) {
      await tapByPointer();
    }
    await new Promise(r => setTimeout(r, 6000));

    out.stepLog.push({
      step: 5,
      ...await readCore(),
    });

    const storageVal = await exec('return localStorage.getItem("hoshikuzu-kobo-v1");');
    const ua = await exec('return navigator.userAgent;');
    out.stepLog.push({ step: 'meta', localStorage: storageVal, navigatorUserAgent: ua });

    const errs = await exec('return window.__consoleErrors || [];');
    out.errors = errs && errs.length ? errs : '無し';

  } catch (e) {
    out.error = String(e.message || e);
  } finally {
    if (out.sessionId) {
      try { await httpJson('DELETE', `${base}/session/${out.sessionId}`); } catch (_) {}
    }
    driver.kill();
  }

  console.log(JSON.stringify(out, null, 2));
})();
```

## 最初の30秒

- 開いた直後は `#amount=0`、`#rate=毎秒 0`、`#goalText=小さな衛星まで 4` でした。
- `#tap` を4回タップすると `#amount` が `4` になりました。
- `#shop` ボタンは3件。
  - ボタン1: `4 と交換所持 0` / 押せる
  - ボタン2: `100 と交換所持 0` / 押せない
  - ボタン3: `800 と交換所持 0` / 押せない
- 最初に押せるボタン（1つ目）を1回押すと `#amount` は `0`、`#rate` は `毎秒 0.2`、`#shop`の所持は衛星1個 (`所持 1`) へ更新。コスト表示も `5 と交換` に変わりました。
- その後 `#tap` を10回し、6秒待つと `#amount=11`、`#rate=毎秒 0.2`、`#goalText=小さな衛星と交換できます` になりました。
- `localStorage` の `hoshikuzu-kobo-v1` は `{"amount":10.943299999999992,"owned":[1,0,0]}`。
- `navigator.userAgent` は `Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) HeadlessChrome/150.0.0.0 Safari/537.36` 。
- コンソールエラーは `無し` でした（今回の実行範囲内）。

## やめる理由

- 1回目の大きな変化（自動化の起点獲得）は早いですが、速度増加は低く、`毎秒 0.2` 程度で伸びが遅いです。
- 途中時点で `#shop` の2番目・3番目がずっとロックされたままで、短時間では「選択肢が限られたまま」になります。
- 初回は数十秒で数字は増えるものの、変化の密度が高くないので、飽きる可能性が高いです。  
  特に `4→5→...` と上がっていく最初の循環は「やることは明確だが報酬比の実感が薄い」タイプです。

## 払うか

払わない。  
同じ作りで払うなら、次が必要です。

- 初回購入時から長い区間で明確に価値が積み上がる設計（`小さな衛星/月面観測所/星間工房` が短時間で到達可能で、各段階の体験が変わること）。
- オフライン進行や進捗の見える化（例えば `localStorage` 以外の保存先、日次の収益通知、段階ボーナス）を入れること。
- 「ただ数を増やす」以外の目的（イベント/ミッション/季節目標/ランキング等）を最低限用意し、プレイ継続の理由を増やすこと。

## 一番弱いところ

- `tapEl.addEventListener('pointerdown', ...)` だけで加算処理が走る構成。`click` でなく `pointerdown` 固有なので、入力方法によっては意図しない挙動になります。
- `setInterval(save, 5000);` と `localStorage` のみ保存（`saveKey = 'hoshikuzu-kobo-v1'`）は、ブラウザクラッシュ時や端末変更時に進捗が消えるリスクが高いです。
- 収益率は `nextCost` と `perSecond` で固定的に増える設計ですが、`small satellite` の `baseCost:4` と `perSecond:0.2`、次価格 `5` への再購入を重ねる初期設計が続くと、序盤の加速度的な達成感が短く、数値の伸びに対して操作体験が単調になりやすいです。
