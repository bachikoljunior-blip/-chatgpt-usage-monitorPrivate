    'use strict';

    const nextCost = (baseCost, owned) => Math.ceil(baseCost * Math.pow(1.15, owned));

    const producerTypes = [
      { name: '小さな衛星', icon: '🛰️', baseCost: 15, perSecond: 0.2 },
      { name: '月面観測所', icon: '🔭', baseCost: 100, perSecond: 1.5 },
      { name: '星間工房', icon: '🚀', baseCost: 800, perSecond: 12 }
    ];
    const saveKey = 'hoshikuzu-kobo-v1';
    const state = { amount: 0, owned: producerTypes.map(() => 0) };

    const amountEl = document.getElementById('amount');
    const rateEl = document.getElementById('rate');
    const shopEl = document.getElementById('shop');
    const tapEl = document.getElementById('tap');
    const goalTextEl = document.getElementById('goalText');
    const goalBarEl = document.getElementById('goalBar');

    const format = value => {
      if (value < 1000) return value.toLocaleString('ja-JP', { maximumFractionDigits: value < 10 ? 1 : 0 });
      const units = [['兆', 1e12], ['億', 1e8], ['万', 1e4]];
      const found = units.find(([, size]) => value >= size);
      return found ? (value / found[1]).toLocaleString('ja-JP', { maximumFractionDigits: 2 }) + found[0] : Math.floor(value).toLocaleString('ja-JP');
    };

    const productionRate = () => producerTypes.reduce((sum, item, index) => sum + item.perSecond * state.owned[index], 0);

    function load() {
      try {
        const saved = JSON.parse(localStorage.getItem(saveKey));
        if (!saved || !Number.isFinite(saved.amount) || !Array.isArray(saved.owned)) return;
        state.amount = Math.max(0, saved.amount);
        state.owned = producerTypes.map((_, index) => Math.max(0, Math.floor(Number(saved.owned[index]) || 0)));
      } catch (_) {
        // 壊れた保存データは無視して、新しいゲームとして始める。
      }
    }

    function save() {
      localStorage.setItem(saveKey, JSON.stringify(state));
    }

    function buildShop() {
      shopEl.replaceChildren(...producerTypes.map((item, index) => {
        const article = document.createElement('article');
        article.className = 'panel producer';
        article.innerHTML = `
          <div class="icon" aria-hidden="true">${item.icon}</div>
          <div><div class="producer-name">${item.name}</div><div class="producer-info">1個で毎秒 ${format(item.perSecond)}</div></div>
          <button class="buy" type="button" data-index="${index}"><span class="cost"></span><span class="owned"></span></button>`;
        return article;
      }));
    }

    function render() {
      amountEl.textContent = format(state.amount);
      rateEl.textContent = `毎秒 ${format(productionRate())}`;
      const buttons = shopEl.querySelectorAll('.buy');
      producerTypes.forEach((item, index) => {
        const cost = nextCost(item.baseCost, state.owned[index]);
        const button = buttons[index];
        button.querySelector('.cost').textContent = `${format(cost)} で購入`;
        button.querySelector('.owned').textContent = `所持 ${state.owned[index]}`;
        button.disabled = state.amount < cost;
        button.setAttribute('aria-label', `${item.name}を星くず${format(cost)}で購入。所持${state.owned[index]}個`);
      });

      const choices = producerTypes.map((item, index) => ({ item, cost: nextCost(item.baseCost, state.owned[index]) }));
      const next = choices.reduce((best, choice) => choice.cost < best.cost ? choice : best);
      goalTextEl.textContent = state.amount >= next.cost ? `${next.item.name}を購入できます` : `${next.item.name}まで ${format(next.cost - state.amount)}`;
      goalBarEl.style.width = `${Math.min(100, state.amount / next.cost * 100)}%`;
    }

    function showPop(clientX, clientY) {
      const pop = document.createElement('span');
      pop.className = 'pop';
      pop.textContent = '+1';
      pop.style.left = `${Math.max(8, Math.min(innerWidth - 42, clientX - 10))}px`;
      pop.style.top = `${Math.max(8, clientY - 15)}px`;
      document.body.append(pop);
      setTimeout(() => pop.remove(), 750);
    }

    tapEl.addEventListener('pointerdown', event => {
      state.amount += 1;
      showPop(event.clientX, event.clientY);
      tapEl.classList.add('pressed');
      render();
      save();
    });
    const releaseTap = () => tapEl.classList.remove('pressed');
    tapEl.addEventListener('pointerup', releaseTap);
    tapEl.addEventListener('pointercancel', releaseTap);
    tapEl.addEventListener('pointerleave', releaseTap);

    shopEl.addEventListener('click', event => {
      const button = event.target.closest('.buy');
      if (!button) return;
      const index = Number(button.dataset.index);
      const item = producerTypes[index];
      const cost = nextCost(item.baseCost, state.owned[index]);
      if (state.amount < cost) return;
      state.amount -= cost;
      state.owned[index] += 1;
      render();
      save();
    });

    load();
    buildShop();
    render();

    let previous = performance.now();
    function tick(now) {
      const elapsedSeconds = Math.min(1, (now - previous) / 1000);
      previous = now;
      state.amount += productionRate() * elapsedSeconds;
      render();
      requestAnimationFrame(tick);
    }
    requestAnimationFrame(tick);
    setInterval(save, 5000);
    addEventListener('pagehide', save);
    document.addEventListener('visibilitychange', () => { if (document.hidden) save(); });
  </script>

const test = require('node:test');

const nextCost = (baseCost, owned) => Math.ceil(baseCost * Math.pow(1.15, owned));

test('0個目は基礎価格そのもの', () => {
  assert.equal(nextCost(15, 0), 15);
});

test('1個所有後は15%上昇して端数を切り上げる', () => {

});

  assert.equal(nextCost(15, 25), 494);
});
