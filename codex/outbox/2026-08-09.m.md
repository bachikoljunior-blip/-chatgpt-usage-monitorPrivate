## 3. コードを書く

`idle-cost.mjs`

```js
export function nextCost(base, growth, owned) {
  if (base <= 0 || growth <= 1 || owned < 0) {
    throw new TypeError("base must be > 0, growth must be > 1, and owned must be >= 0");
  }

  return Math.ceil(base * growth ** owned);
}

export function totalCost(base, growth, from, to) {
  if (from < 0 || to < from) {
    throw new TypeError("from must be >= 0 and to must be >= from");
  }

  let total = 0;
  for (let owned = from; owned < to; owned += 1) {
    total += nextCost(base, growth, owned);
  }
  return total;
}
```

`idle-cost.test.mjs`

```js
import assert from "node:assert/strict";
import test from "node:test";
import { nextCost, totalCost } from "./idle-cost.mjs";

test("owned = 0 returns the base price", () => {
  assert.equal(nextCost(10, 1.5, 0), 10);
});

test("a fractional price is rounded up", () => {
  assert.equal(nextCost(10, 1.15, 1), 12);
});

test("totalCost uses an inclusive lower and exclusive upper bound", () => {
  assert.equal(totalCost(10, 2, 0, 3), 70);
});

test("an empty range costs zero", () => {
  assert.equal(totalCost(10, 2, 3, 3), 0);
});

test("base = 0 throws TypeError", () => {
  assert.throws(() => nextCost(0, 2, 0), TypeError);
});

test("growth = 1 throws TypeError", () => {
  assert.throws(() => nextCost(10, 1, 0), TypeError);
});

test("negative owned throws TypeError", () => {
  assert.throws(() => nextCost(10, 2, -1), TypeError);
});

test("a reversed totalCost range throws TypeError", () => {
  assert.throws(() => totalCost(10, 2, 3, 2), TypeError);
});
```

実行コマンド:

```sh
node --test idle-cost.test.mjs
```

目標は月額 `¥200,000`、商品価格は `$25.00` ですが、`state/eta.json` の `fx_usd_jpy_assumed` は `null` です。円建て目標をドル建て単価で割るための為替レートが、貼り付けられたファイルにはありません。

したがって、ファイル外の為替を仮定せずに必要販売本数を確定できません。USD/JPYを `r` 円／ドルと置く場合の式は次のとおりです。

```text
必要本数 = ceil(200,000 / (25 × r))
```

なお、手数料や税引後の目標であるとは記載されていないため、それらを加味した本数も算出できません。
